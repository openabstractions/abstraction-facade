cmake_minimum_required(VERSION 3.16)
project(abstraction_facade VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(ABSTRACTION_FACADE_BUILD_AGGREGATE "Build convenience client for logging/config/router/jobs" ON)
option(ABSTRACTION_FACADE_BUILD_JOBS "Build only the job service binding when aggregate is disabled" OFF)
option(ABSTRACTION_FACADE_BUILD_RESOLUTION "Build independent resolver binding" OFF)
option(ABSTRACTION_FACADE_BUILD_LOGGING "Build independent logging binding" OFF)
option(ABSTRACTION_FACADE_BUILD_ASKS "Build independent question binding" OFF)
option(ABSTRACTION_FACADE_BUILD_RIGHTS "Build independent rights binding" OFF)
option(ABSTRACTION_FACADE_BUILD_STORAGE "Build independent storage binding" OFF)
add_library(abstraction_facade_protocol INTERFACE)
add_library(abstraction::facade_protocol ALIAS abstraction_facade_protocol)
set_target_properties(abstraction_facade_protocol PROPERTIES EXPORT_NAME facade_protocol)
target_compile_features(abstraction_facade_protocol INTERFACE cxx_std_17)
target_include_directories(abstraction_facade_protocol INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
install(TARGETS abstraction_facade_protocol EXPORT abstraction_facade_resolutionTargets)
install(DIRECTORY abstraction/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/abstraction)
set(facade_has_resolution OFF)
if(ABSTRACTION_FACADE_BUILD_AGGREGATE OR ABSTRACTION_FACADE_BUILD_JOBS OR ABSTRACTION_FACADE_BUILD_RESOLUTION OR ABSTRACTION_FACADE_BUILD_STORAGE OR ABSTRACTION_FACADE_BUILD_ASKS OR ABSTRACTION_FACADE_BUILD_RIGHTS OR ABSTRACTION_FACADE_BUILD_LOGGING)
    set(facade_has_resolution ON)
    if(NOT TARGET abstraction::ipc)
        find_package(abstraction_ipc CONFIG QUIET)
    endif()
    if(NOT TARGET abstraction::ipc AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../abstraction-identity/cpp/CMakeLists.txt")
        add_subdirectory(../../abstraction-identity/cpp abstraction_ipc-build)
    endif()
    if(NOT TARGET abstraction::ipc)
        message(FATAL_ERROR "Install abstraction_ipc or provide sibling abstraction-identity. Nothing is fetched.")
    endif()
    add_library(abstraction_facade_resolution INTERFACE)
    add_library(abstraction::facade_resolution ALIAS abstraction_facade_resolution)
    set_target_properties(abstraction_facade_resolution PROPERTIES EXPORT_NAME facade_resolution)
    target_include_directories(abstraction_facade_resolution INTERFACE
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    target_link_libraries(abstraction_facade_resolution INTERFACE abstraction::facade_protocol abstraction::ipc)
    install(TARGETS abstraction_facade_resolution EXPORT abstraction_facade_resolutionTargets)
    install(FILES include/abstraction/facade/resolution.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/abstraction/facade)
endif()
if(ABSTRACTION_FACADE_BUILD_AGGREGATE OR ABSTRACTION_FACADE_BUILD_JOBS)
    if(NOT TARGET abstraction::job_acceptance)
        find_package(abstraction_job_acceptance CONFIG QUIET)
    endif()
    if(NOT TARGET abstraction::job_acceptance AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../abstraction-job/cpp/CMakeLists.txt")
        # Limit this dependency to generated API. Do not build the embedded store.
        set(ABSTRACTION_JOB_BUILD_LEGACY OFF)
        add_subdirectory(../../abstraction-job/cpp abstraction_job_acceptance-build)
        unset(ABSTRACTION_JOB_BUILD_LEGACY)
    endif()
    if(NOT TARGET abstraction::job_acceptance)
        message(FATAL_ERROR "Install abstraction_job_acceptance or provide sibling abstraction-job. Nothing is fetched.")
    endif()
    add_library(abstraction_facade_jobs INTERFACE)
    add_library(abstraction::facade_jobs ALIAS abstraction_facade_jobs)
    set_target_properties(abstraction_facade_jobs PROPERTIES EXPORT_NAME facade_jobs)
    target_include_directories(abstraction_facade_jobs INTERFACE
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    target_link_libraries(abstraction_facade_jobs INTERFACE abstraction::facade_resolution abstraction::job_acceptance)
    install(TARGETS abstraction_facade_jobs EXPORT abstraction_facadeTargets)
    install(FILES include/abstraction/facade/inventory.hpp include/abstraction/facade/jobs.hpp
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/abstraction/facade)
endif()
if(ABSTRACTION_FACADE_BUILD_AGGREGATE)
foreach(capability logging config router model)
    if(NOT TARGET abstraction::${capability}_client)
        find_package(abstraction_${capability} CONFIG QUIET)
    endif()
    if(NOT TARGET abstraction::${capability}_client AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../abstraction-${capability}/cpp/CMakeLists.txt")
        add_subdirectory(../../abstraction-${capability}/cpp abstraction_${capability}-build)
    endif()
    if(NOT TARGET abstraction::${capability}_client)
        message(FATAL_ERROR "Install abstraction_${capability} and set CMAKE_PREFIX_PATH, or provide sibling abstraction-${capability}. Nothing is fetched.")
    endif()
endforeach()
add_library(abstraction_facade_client INTERFACE)
add_library(abstraction::facade_client ALIAS abstraction_facade_client)
set_target_properties(abstraction_facade_client PROPERTIES EXPORT_NAME facade_client)
target_compile_features(abstraction_facade_client INTERFACE cxx_std_17)
target_include_directories(abstraction_facade_client INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(abstraction_facade_client INTERFACE abstraction::facade_jobs abstraction::logging_client abstraction::config_client abstraction::router_client abstraction::model_client)
install(TARGETS abstraction_facade_client EXPORT abstraction_facadeTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} PATTERN "storage.hpp" EXCLUDE PATTERN "asks.hpp" EXCLUDE PATTERN "rights.hpp" EXCLUDE)
endif()
if(ABSTRACTION_FACADE_BUILD_STORAGE)
    if(NOT TARGET abstraction::storage_client)
        find_package(abstraction_storage_content CONFIG QUIET)
    endif()
    if(NOT TARGET abstraction::storage_client AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../abstraction-storage/cpp/CMakeLists.txt")
        set(ABSTRACTION_STORAGE_BUILD_SERVICE ON)
        set(ABSTRACTION_STORAGE_BUILD_LEGACY OFF)
        add_subdirectory(../../abstraction-storage/cpp abstraction_storage-build)
    endif()
    if(NOT TARGET abstraction::storage_client)
        message(FATAL_ERROR "Install abstraction_storage_content or provide sibling abstraction-storage. Nothing is fetched.")
    endif()
    add_library(abstraction_facade_storage INTERFACE)
    add_library(abstraction::facade_storage ALIAS abstraction_facade_storage)
    set_target_properties(abstraction_facade_storage PROPERTIES EXPORT_NAME facade_storage)
    target_link_libraries(abstraction_facade_storage INTERFACE abstraction::facade_resolution abstraction::storage_client)
    install(TARGETS abstraction_facade_storage EXPORT abstraction_facade_storageTargets)
    install(FILES include/abstraction/facade/storage.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/abstraction/facade)
endif()
foreach(capability asks rights logging)
    string(TOUPPER "${capability}" upper)
    if(ABSTRACTION_FACADE_BUILD_${upper})
        if(NOT TARGET abstraction::${capability}_client)
            find_package(abstraction_${capability} CONFIG QUIET)
        endif()
        if(NOT TARGET abstraction::${capability}_client AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../abstraction-${capability}/cpp/CMakeLists.txt")
            add_subdirectory(../../abstraction-${capability}/cpp abstraction_${capability}-build)
        endif()
        if(NOT TARGET abstraction::${capability}_client)
            message(FATAL_ERROR "Install abstraction_${capability}. Nothing is fetched.")
        endif()
        add_library(abstraction_facade_${capability} INTERFACE)
        add_library(abstraction::facade_${capability} ALIAS abstraction_facade_${capability})
        set_target_properties(abstraction_facade_${capability} PROPERTIES EXPORT_NAME facade_${capability})
        target_link_libraries(abstraction_facade_${capability} INTERFACE abstraction::facade_resolution abstraction::${capability}_client)
        install(TARGETS abstraction_facade_${capability} EXPORT abstraction_facade_${capability}Targets)
        install(FILES include/abstraction/facade/${capability}.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/abstraction/facade)
    endif()
endforeach()
foreach(package resolution storage asks rights logging)
    string(TOUPPER "${package}" upper)
    if(NOT package STREQUAL "resolution" AND NOT ABSTRACTION_FACADE_BUILD_${upper})
        continue()
    endif()
    set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/abstraction_facade_${package})
    install(EXPORT abstraction_facade_${package}Targets NAMESPACE abstraction:: DESTINATION ${package_dir})
    configure_package_config_file(cmake/abstraction_facade_${package}Config.cmake.in
        ${CMAKE_CURRENT_BINARY_DIR}/abstraction_facade_${package}Config.cmake INSTALL_DESTINATION ${package_dir})
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/abstraction_facade_${package}Config.cmake DESTINATION ${package_dir})
endforeach()
set(config_dir ${CMAKE_INSTALL_LIBDIR}/cmake/abstraction_facade)
if(ABSTRACTION_FACADE_BUILD_AGGREGATE OR ABSTRACTION_FACADE_BUILD_JOBS)
install(EXPORT abstraction_facadeTargets NAMESPACE abstraction:: DESTINATION ${config_dir})
endif()
configure_package_config_file(cmake/abstraction_facadeConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/abstraction_facadeConfig.cmake INSTALL_DESTINATION ${config_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/abstraction_facadeConfigVersion.cmake COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/abstraction_facadeConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/abstraction_facadeConfigVersion.cmake DESTINATION ${config_dir})
