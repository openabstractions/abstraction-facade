cmake_minimum_required(VERSION 3.16)
project(facade_jobs_consumer LANGUAGES CXX)
find_package(abstraction_facade CONFIG REQUIRED)
if(TARGET abstraction::logging_client OR TARGET abstraction::config_client OR TARGET abstraction::router_client OR TARGET abstraction::job)
    message(FATAL_ERROR "Job-only binding pulled unrelated clients or an embedded store")
endif()
add_executable(facade_jobs_consumer main.cpp)
target_link_libraries(facade_jobs_consumer PRIVATE abstraction::facade_jobs)
enable_testing()
add_test(NAME facade_jobs_consumer COMMAND facade_jobs_consumer)
