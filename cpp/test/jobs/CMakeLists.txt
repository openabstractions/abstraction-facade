cmake_minimum_required(VERSION 3.16)
project(facade_jobs_consumer LANGUAGES CXX)
find_package(abstraction_facade CONFIG REQUIRED)
option(OA_TEST_DOWNLOAD_REQUEST "Exercise typed HTTP requests through the installed request API" OFF)
if(OA_TEST_DOWNLOAD_REQUEST)
    find_package(abstraction_download_request CONFIG REQUIRED)
endif()
if(TARGET abstraction::logging_client OR TARGET abstraction::config_client OR TARGET abstraction::router_client OR TARGET abstraction::job OR TARGET abstraction::download_runner OR TARGET abstraction::download)
    message(FATAL_ERROR "Job-only binding pulled unrelated clients or an embedded store")
endif()
add_executable(facade_jobs_consumer main.cpp)
target_link_libraries(facade_jobs_consumer PRIVATE abstraction::facade_jobs)
if(OA_TEST_DOWNLOAD_REQUEST)
    target_link_libraries(facade_jobs_consumer PRIVATE abstraction::download_request)
    target_compile_definitions(facade_jobs_consumer PRIVATE OA_DOWNLOAD_REQUEST=1)
endif()
enable_testing()
add_test(NAME facade_jobs_consumer COMMAND facade_jobs_consumer)
